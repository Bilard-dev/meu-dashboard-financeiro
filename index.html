async function loadDashboardData() { try { // BUSCA DIRETA NO SUPABASE FILTRANDO PELO USUÁRIO const { data, error } = await _supabase .from('lancamentos') .select('*') .eq('user_id', currentUser) .order('created_at', { ascending: false }); if (error) throw error; globalData = data.map(item => { // 1. TRATAMENTO INTELIGENTE DA DATA let dateObj = new Date(); if (item.data) { const strData = String(item.data).trim(); if (strData.includes('/')) { // Formato Brasileiro: DD/MM/YYYY const p = strData.split('/'); if (p.length === 3) { dateObj = new Date(parseInt(p[2], 10), parseInt(p[1], 10) - 1, parseInt(p[0], 10)); } } else if (strData.includes('-')) { // Formato ISO: YYYY-MM-DD const p = strData.split('-'); if (p.length === 3) { dateObj = new Date(parseInt(p[0], 10), parseInt(p[1], 10) - 1, parseInt(p[2], 10)); } } } // 2. TRATAMENTO INTELIGENTE DO VALOR (Limpa R$, espaços e converte vírgula para ponto) let valNum = 0; if (item.valor !== null && item.valor !== undefined) { let valStr = String(item.valor) .replace('R$', '') .replace(/\s/g, '') .replace(/\./g, '') // remove ponto de milhar se houver .replace(',', '.'); // troca vírgula decimal por ponto valNum = parseFloat(valStr) || 0; } return { id: item.id, type: String(item.tipo || 'DESPESA').toUpperCase().trim(), date: isNaN(dateObj.getTime()) ? new Date() : dateObj, desc: item.descricao || '', custo: item.custo || '', category: item.categoria || 'Outros', subCat: item.subcategoria || '', value: valNum, pagamento: item.pagamento || '', cartao: item.cartao || '', parcela: item.parcela || '', month: isNaN(dateObj.getTime()) ? new Date().getMonth() : dateObj.getMonth(), year: isNaN(dateObj.getTime()) ? new Date().getFullYear() : dateObj.getFullYear() }; }); document.getElementById('loading').style.display = 'none'; document.getElementById('dashboard').style.display = 'block'; setupMonthSelector(); setupInvestSelector(); updateDashboard(document.getElementById('monthSelector').value); } catch (error) { console.error(error); document.getElementById('loading').innerText = "Erro ao carregar dados do Supabase: " + error.message; } }
